CREATE TABLE Publisher (
    Publisher_ID INT PRIMARY KEY,
    Name VARCHAR(100)
);

CREATE TABLE Category (
    Category_ID INT PRIMARY KEY,
    Name VARCHAR(50)
);

CREATE TABLE Library_Branch (
    Branch_ID INT PRIMARY KEY,
    Branch_Name VARCHAR(100),
    Address VARCHAR(200)
);

CREATE TABLE Book (
    Book_ID INT PRIMARY KEY,
    Title VARCHAR(255),
    ISBN VARCHAR(20),
    Publisher_ID INT,
    Category_ID INT,
    Branch_ID INT,
    FOREIGN KEY (Publisher_ID) REFERENCES Publisher(Publisher_ID),
    FOREIGN KEY (Category_ID) REFERENCES Category(Category_ID),
    FOREIGN KEY (Branch_ID) REFERENCES Library_Branch(Branch_ID)
);

CREATE TABLE Author (
    Author_ID INT PRIMARY KEY,
    Name VARCHAR(100)
);

CREATE TABLE Book_Author (
    Book_ID INT,
    Author_ID INT,
    PRIMARY KEY (Book_ID, Author_ID),
    FOREIGN KEY (Book_ID) REFERENCES Book(Book_ID),
    FOREIGN KEY (Author_ID) REFERENCES Author(Author_ID)
);

CREATE TABLE Member (
    Member_ID INT PRIMARY KEY,
    Name VARCHAR(100),
    Contact VARCHAR(20),
    Email VARCHAR(100),
    Branch_ID INT,
    FOREIGN KEY (Branch_ID) REFERENCES Library_Branch(Branch_ID)
);

CREATE TABLE Staff (
    Staff_ID INT PRIMARY KEY,
    Name VARCHAR(100),
    Email VARCHAR(100),
    Branch_ID INT,
    FOREIGN KEY (Branch_ID) REFERENCES Library_Branch(Branch_ID)
);

CREATE TABLE Transaction (
    Transaction_ID INT PRIMARY KEY,
    Book_ID INT,
    Member_ID INT,
    Staff_ID INT,
    Issue_Date DATE,
    Return_Date DATE,
    FOREIGN KEY (Book_ID) REFERENCES Book(Book_ID),
    FOREIGN KEY (Member_ID) REFERENCES Member(Member_ID),
    FOREIGN KEY (Staff_ID) REFERENCES Staff(Staff_ID)
);

CREATE TABLE Fine (
    Fine_ID INT PRIMARY KEY,
    Transaction_ID INT,
    Amount DECIMAL(6,2),
    Status VARCHAR(20),
    FOREIGN KEY (Transaction_ID) REFERENCES Transaction(Transaction_ID)
);

CREATE TABLE Reservation (
    Reservation_ID INT PRIMARY KEY,
    Book_ID INT,
    Member_ID INT,
    Staff_ID INT,
    Reservation_Date DATE,
    FOREIGN KEY (Book_ID) REFERENCES Book(Book_ID),
    FOREIGN KEY (Member_ID) REFERENCES Member(Member_ID),
    FOREIGN KEY (Staff_ID) REFERENCES Staff(Staff_ID)
);

SELECT * FROM Author;

SELECT B.Title, A.Name AS Author_Name
FROM Book B
JOIN Book_Author BA ON B.Book_ID = BA.Book_ID
JOIN Author A ON BA.Author_ID = A.Author_ID;

SELECT * FROM Member;

SELECT T.Transaction_ID, M.Name AS Member, B.Title AS Book, T.Issue_Date, T.Return_Date
FROM Transaction T
JOIN Member M ON T.Member_ID = M.Member_ID
JOIN Book B ON T.Book_ID = B.Book_ID;

SELECT T.Transaction_ID, M.Name, B.Title, T.Issue_Date
FROM Transaction T
JOIN Member M ON T.Member_ID = M.Member_ID
JOIN Book B ON T.Book_ID = B.Book_ID
WHERE T.Return_Date IS NULL;

SELECT F.Fine_ID, M.Name AS Member, B.Title AS Book, F.Amount, F.Status
FROM Fine F
JOIN Transaction T ON F.Transaction_ID = T.Transaction_ID
JOIN Member M ON T.Member_ID = M.Member_ID
JOIN Book B ON T.Book_ID = B.Book_ID;

SELECT R.Reservation_ID, M.Name AS Member, B.Title AS Book, R.Reservation_Date
FROM Reservation R
JOIN Member M ON R.Member_ID = M.Member_ID
JOIN Book B ON R.Book_ID = B.Book_ID;
